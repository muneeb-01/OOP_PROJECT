class Student {

    String name;
    int rollNo;

    Student(String name, int rollNo) {
        this.name = name;
        this.rollNo = rollNo;
    }

    void display() {
        System.out.println("Name: " + name);
        System.out.println("Roll No: " + rollNo);
    }
}

class UndergraduateStudent extends Student {

    int semester;

    UndergraduateStudent(String name, int rollNo, int semester) {
        super(name, rollNo);
        this.semester = semester;
    }

    void display() {
        super.display();
        System.out.println("Semester: " + semester);
    }
}

class GraduateStudent extends Student {

    String researchTopic;

    GraduateStudent(String name, int rollNo, String researchTopic) {
        super(name, rollNo);
        this.researchTopic = researchTopic;
    }

    void display() {
        super.display();
        System.out.println("Research Topic: " + researchTopic);
    }
}

public class Main {
    public static void main(String[] args) {

        UndergraduateStudent s1 =
                new UndergraduateStudent("Ali", 101, 5);

        GraduateStudent s2 =
                new GraduateStudent("Ahmed", 102, "Artificial Intelligence");

        s1.display();

        System.out.println();

        s2.display();
    }
}
